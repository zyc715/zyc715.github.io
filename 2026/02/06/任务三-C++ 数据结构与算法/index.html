一、STL容器

1. 序列式容器（线性结构）

vector（动态数组） 连续存储 | 随机访问O(1) | 自动扩容

#include <iostream>
#include <vector>
using namespace std;
int main() {
    vector<int> v1;
    vector<int> v2(5,10);
    vector<int> v3(v2.begin(),v2.end());
    vector<int> v4={1,2,3,4,5};
    v4.push_back(6);
    v4.pop_back();
    v4.insert(v4.begin()+2,9);
    v4.erase(v4.begin()+2);
    v4.resize(8,0);
    v4.clear();
    for(int i=0;i<v2.size();i++) cout<<v2[i]<<" ";
    for(auto it=v2.begin();it!=v2.end();it++) cout<<*it<<" ";
    for(int num:v2) cout<<num<<" ";
    cout<<"大小: "<<v2.size()<<endl;
    cout<<"容量: "<<v2.capacity()<<endl;
    cout<<"是否为空: "<<v2.empty()<<endl;
    return 0;
}

#include <iostream>
#include <list>
using namespace std;
int main() {
    list<int> l1;

    l3.push_front(0);
    l3.push_back(6);
    l3.pop_front();
    l3.pop_back();
    auto it=l3.begin();
    advance(it,2);
    l3.insert(it,9);

二、数据结构

1. 线性结构

单链表 遍历/插入/删除 | 动态内存管理

#include <iostream>
using namespace std;
struct ListNode {
    int val;
    ListNode* next;
    ListNode(int x):val(x),next(nullptr){}
};
void traverse(ListNode* head) {
    ListNode* curr=head;
    while(curr) {
        cout<<curr->val<<" ";
        curr=curr->next;
    }
}
void insertNode(ListNode*& head,int val) {
    ListNode* newNode=new ListNode(val);
    newNode->next=head->next;
    head->next=newNode;
}
void deleteNode(ListNode*& head,int val) {
    ListNode* curr=head;
    while(curr->next&&curr->next->val!=val) curr=curr->next;
    if(curr->next) {
        ListNode* temp=curr->next;
        curr->next=curr->next->next;
        delete temp;
    }
}
int main() {
    ListNode* head=new ListNode(0);
    insertNode(head,1);
    insertNode(head,2);
    traverse(head->next);
    deleteNode(head,1);
    traverse(head->next);
    return 0;
}

栈/队列（数组实现） 栈：后进先出 | 队列：循环数组避免假溢出

#include <iostream>
using namespace std;
const int MAX_SIZE=100;
struct Stack {
    int data[MAX_SIZE];
    int top;
    Stack():top(-1){}
    bool push(int val) {
        if(top==MAX_SIZE-1) return false;
        data[++top]=val;
        return true;
    }
    bool pop() {
        if(top==-1) return false;
        top--;
        return true;
    }
    int topVal() {return data[top];}
    bool empty() {return top==-1;}
};
struct Queue {
    int data[MAX_SIZE];
    int front,rear;
    Queue():front(0),rear(0){}
    bool push(int val) {
        if((rear+1)%MAX_SIZE==front) return false;
        data[rear]=val;
        rear=(rear+1)%MAX_SIZE;
        return true;
    }
    bool pop() {
        if(front==rear) return false;
        front=(front+1)%MAX_SIZE;
        return true;
    }
    int frontVal() {return data[front];}
    bool empty() {return front==rear;}
};
int main() {
    Stack st;
    st.push(1);st.push(2);
    cout<<st.topVal()<<endl;
    Queue q;
    q.push(1);q.push(2);
    cout<<q.frontVal()<<endl;
    return 0;
}

2. 非线性结构

二叉搜索树 插入/查找/中序遍历 | 升序输出

#include <iostream>
using namespace std;
struct TreeNode {
    int val;
    TreeNode* left;
    TreeNode* right;
    TreeNode(int x):val(x),left(nullptr),right(nullptr){}
};
TreeNode* insert(TreeNode* root,int val) {
    if(!root) return new TreeNode(val);
    if(val<root->val) root->left=insert(root->left,val);
    else root->right=insert(root->right,val);
    return root;
}
bool search(TreeNode* root,int val) {
    if(!root) return false;
    if(val==root->val) return true;
    return val<root->val?search(root->left,val):search(root->right,val);
}
void inorder(TreeNode* root) {
    if(root) {
        inorder(root->left);
        cout<<root->val<<" ";
        inorder(root->right);
    }
}
int main() {
    TreeNode* root=nullptr;
    root=insert(root,5);
    insert(root,3);insert(root,7);insert(root,2);
    inorder(root);
    cout<<search(root,3)<<endl;
    return 0;
}

堆 & 哈希表 堆排序 | 哈希表链地址法解决冲突

#include <iostream>
#include <vector>
#include <string>
using namespace std;
void heapify(vector<int>& arr,int n,int i) {
    int largest=i;
    int left=2*i+1;
    int right=2*i+2;
    if(left<n&&arr[left]>arr[largest]) largest=left;
    if(right<n&&arr[right]>arr[largest]) largest=right;
    if(largest!=i) {
        swap(arr[i],arr[largest]);
        heapify(arr,n,largest);
    }
}
void heapSort(vector<int>& arr) {
    int n=arr.size();
    for(int i=n/2-1;i>=0;i--) heapify(arr,n,i);
    for(int i=n-1;i>0;i--) {
        swap(arr[0],arr[i]);
        heapify(arr,i,0);
    }
}
struct HashNode {
    string key;
    int val;
    HashNode* next;
    HashNode(string k,int v):key(k),val(v),next(nullptr){}
};
const int TABLE_SIZE=100;
class HashTable {
private:
    vector<HashNode*> table;
    int hashFunc(string key) {
        int hash=0;
        for(char c:key) hash=(hash*31+c)%TABLE_SIZE;
        return hash;
    }
public:
    HashTable():table(TABLE_SIZE,nullptr){}
    void insert(string key,int val) {
        int idx=hashFunc(key);
        HashNode* newNode=new HashNode(key,val);
        newNode->next=table[idx];
        table[idx]=newNode;
    }
    int search(string key) {
        int idx=hashFunc(key);
        HashNode* curr=table[idx];
        while(curr) {
            if(curr->key==key) return curr->val;
            curr=curr->next;
        }
        return -1;
    }
    void erase(string key) {
        int idx=hashFunc(key);
        HashNode* curr=table[idx];
        HashNode* prev=nullptr;
        while(curr&&curr->key!=key) {
            prev=curr;
            curr=curr->next;
        }
        if(!curr) return;
        if(!prev) table[idx]=curr->next;
        else prev->next=curr->next;
        delete curr;
    }
};
int main() {
    vector<int> arr={4,10,3,5,1};
    heapSort(arr);
    for(int num:arr) cout<<num<<" ";
    HashTable ht;
    ht.insert("apple",5);
    cout<<ht.search("apple")<<endl;
    return 0;
}

三、算法

1. 排序算法

快速排序 分治 | 原地排序 | 效率高

#include <iostream>
#include <vector>
using namespace std;
int partition(vector<int>& arr,int low,int high) {
    int pivot=arr[high];
    int i=low-1;
    for(int j=low;j<high;j++) {
        if(arr[j]<=pivot) swap(arr[++i],arr[j]);
    }
    swap(arr[i+1],arr[high]);
    return i+1;
}
void quickSort(vector<int>& arr,int low,int high) {
    if(low<high) {
        int pi=partition(arr,low,high);
        quickSort(arr,low,pi-1);
        quickSort(arr,pi+1,high);
    }
}
int main() {
    vector<int> arr={64,34,25,12,22,11,90};
    quickSort(arr,0,arr.size()-1);
    for(int num:arr) cout<<num<<" ";
    return 0;
}

复杂度：平均O(n log n) | 最坏O(n²) | 不稳定 | 原地排序

归并排序 分治 | 稳定 | 非原地

#include <iostream>
#include <vector>
using namespace std;
void merge(vector<int>& arr,int low,int mid,int high) {
    vector<int> temp(high-low+1);
    int i=low,j=mid+1,k=0;
    while(i<=mid&&j<=high) temp[k++]=arr[i]<=arr[j]?arr[i++]:arr[j++];
    while(i<=mid) temp[k++]=arr[i++];
    while(j<=high) temp[k++]=arr[j++];
    for(k=0;k<temp.size();k++) arr[low+k]=temp[k];
}
void mergeSort(vector<int>& arr,int low,int high) {
    if(low<high) {
        int mid=low+(high-low)/2;
        mergeSort(arr,low,mid);
        mergeSort(arr,mid+1,high);
        merge(arr,low,mid,high);
    }
}
int main() {
    vector<int> arr={38,27,43,3,9,82,10};
    mergeSort(arr,0,arr.size()-1);
    for(int num:arr) cout<<num<<" ";
    return 0;
}

复杂度：O(n log n) | 稳定 | 非原地排序 | 空间复杂度O(n)

2. 查找/动态规划/递归

二分查找 + 动态规划（0-1背包） + 递归

#include <iostream>
#include <vector>
using namespace std;
int binarySearch(vector<int>& arr,int target) {
    int left=0,right=arr.size()-1;
    while(left<=right) {
        int mid=left+(right-left)/2;
        if(arr[mid]==target) return mid;
        else if(arr[mid]<target) left=mid+1;
        else right=mid-1;
    }
    return -1;
}
int knapsack(vector<int>& weights,vector<int>& values,int capacity) {
    int n=weights.size();
    vector<vector<int>> dp(n+1,vector<int>(capacity+1,0));
    for(int i=1;i<=n;i++) {
        for(int j=1;j<=capacity;j++) {
            if(j>=weights[i-1]) {
                dp[i][j]=max(dp[i-1][j],dp[i-1][j-weights[i-1]]+values[i-1]);
            } else {
                dp[i][j]=dp[i-1][j];
            }
        }
    }
    return dp[n][capacity];
}
int fibRecursion(int n) {
    if(n<=1) return n;
    return fibRecursion(n-1)+fibRecursion(n-2);
}
int main() {
    vector<int> arr={1,3,5,7,9,11,13};
    cout<<binarySearch(arr,7)<<endl;
    vector<int> weights={2,3,4,5};
    vector<int> values={3,4,5,6};
    cout<<knapsack(weights,values,8)<<endl;
    cout<<fibRecursion(10)<<endl;
    return 0;
}

3. 图算法（DFS/BFS）

邻接矩阵 + 深度优先/广度优先搜索

#include <iostream>
#include <vector>
#include <queue>
using namespace std;
class Graph {
private:
    int n;
    vector<vector<int>> adj;
public:
    Graph(int num):n(num),adj(num,vector<int>(num,0)){}
    void addEdge(int u,int v) {
        adj[u][v]=1;
        adj[v][u]=1;
    }
    void dfs(int u,vector<bool>& visited) {
        visited[u]=true;
        cout<<u<<" ";
        for(int v=0;v<n;v++) {
            if(adj[u][v]==1&&!visited[v]) {
                dfs(v,visited);
            }
        }
    }
    void bfs(int start) {
        vector<bool> visited(n,false);
        queue<int> q;
        q.push(start);
        visited[start]=true;
        while(!q.empty()) {
            int u=q.front();
            q.pop();
            cout<<u<<" ";
            for(int v=0;v<n;v++) {
                if(adj[u][v]==1&&!visited[v]) {
                    q.push(v);
                    visited[v]=true;
                }
            }
        }
    }
};
int main() {
    Graph g(5);
    g.addEdge(0,1);g.addEdge(0,2);
    g.addEdge(1,3);g.addEdge(2,4);
    cout<<"DFS: ";
    vector<bool> visited(5,false);
    g.dfs(0,visited);
    cout<<"\nBFS: ";
    g.bfs(0);
    return 0;
}
